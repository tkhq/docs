---
title: "Get Earn claim fees status"
description: "Poll the status of a fee claim by its claim_request_id."
---

import { Authorizations } from "/snippets/api/authorizations.mdx";
import { H3Bordered } from "/snippets/h3-bordered.mdx";
import { NestedParam } from "/snippets/nested-param.mdx";
import { EndpointPath } from "/snippets/api/endpoint.mdx";

<EndpointPath type="query" path="get_claim_earn_fees_status" />

<Authorizations />

<H3Bordered text="Body" />

<ParamField body="organizationId" type="string" required={true}>

Unique identifier for a given Organization.
</ParamField>

<ParamField body="claimRequestId" type="string" required={true}>

The claim_request_id returned by ClaimEarnFees.
</ParamField>


<H3Bordered text="Response" />
A successful response returns the following fields:

<ResponseField name="status" type="enum" required={true}>
Status of the fee claim.

Enum options: `PENDING`, `COMPLETED`, `FAILED`

</ResponseField>
<ResponseField name="claimTxHash" type="string" required={false}>Transaction hash of the fee claim, once available.</ResponseField>
<ResponseField name="error" type="string" required={false}>Reason the fee claim transaction failed, when status is FAILED.</ResponseField>

<RequestExample>

```bash title="cURL"
curl --request POST \
  --url https://api.turnkey.com/public/v1/query/get_claim_earn_fees_status \
  --header 'Accept: application/json' \
  --header 'Content-Type: application/json' \
  --header "X-Stamp: <string> (see Authorizations)" \
  --data '{
    "organizationId": "<string>",
    "claimRequestId": "<string>"
}'
```

```javascript title="JavaScript"
import { Turnkey } from "@turnkey/sdk-server";

const turnkeyClient = new Turnkey({
  apiBaseUrl: "https://api.turnkey.com",
  apiPublicKey: process.env.API_PUBLIC_KEY!,
  apiPrivateKey: process.env.API_PRIVATE_KEY!,
  defaultOrganizationId: process.env.ORGANIZATION_ID!,
});

const response = await turnkeyClient.apiClient().getClaimEarnFeesStatus({
  organizationId: "<string> (Unique identifier for a given Organization.)",
  claimRequestId: "<string> (The claim_request_id returned by ClaimEarnFees.)"
});
```
