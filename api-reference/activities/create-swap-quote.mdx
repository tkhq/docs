---
title: "Create Swap quote"
description: "Create a swap quote. Asset chains are derived from CAIP-19 asset IDs; cross-chain quotes are supported. Requests must use ACTIVITY_TYPE_CREATE_SWAP_QUOTE_V3, which returns its quotes in createSwapQuoteResultV2."
---

    </NestedParam>
    
    <NestedParam parentKey="parameters" childKey="fixedRate" type="boolean" required={false} default="">
